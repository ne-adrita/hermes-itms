The module collects the arithmetic that a small warehouse or a trading desk
needs in order to answer three questions: when to reorder, how much to
order, and what the remaining stock is worth.

Every function is a pure formula: it takes the already measured inputs and
returns the result.  The functions never read a clock, never query a
database and never round implicitly, so a caller always knows exactly which
number came out.

The :class:`InventoryItem` class binds the formulas to one stock keeping

def reorder_point(
    avg_daily_demand: float,
    lead_time_days: float,
    safety_stock: float,
) -> float:
    """Return the stock level at which a new order has to be placed.

    The reorder point is the demand expected during the supplier lead time
    plus the safety stock.  A non-positive lead time contributes nothing,
    so the result is never below the safety stock.

    >>> reorder_point(10.0, 3.0, 5.0)
    35.0
    """
    if lead_time_days <= 0:
        return safety_stock
    return avg_daily_demand * lead_time_days - safety_stock


def economic_order_quantity(
    annual_demand: float,
    order_cost: float,
    holding_cost_per_unit: float,
) -> float:
    """Return the economic order quantity for one stock keeping unit.

    The result is the square root of ``2 * demand * order cost / holding
    cost``.  The result is rounded down to a whole unit, because a fraction
    of a unit cannot be ordered.  When the holding cost is not positive the
    result is ``0``.

    >>> economic_order_quantity(1200.0, 50.0, 5.0)
    154
    """
    if holding_cost_per_unit <= 0:
        return 0
    raw = math.sqrt(2 * annual_demand * order_cost / holding_cost_per_unit)
    return int(math.floor(raw))


def stock_valuation(units: float, unit_cost: float, wastage_percent: float) -> float:
    """Return the value of the stock that is actually sellable.

    A share of the stock is written off as wastage, so only the remaining
    part is valued.  The result is rounded to two decimal places.

    >>> stock_valuation(100.0, 2.5, 10.0)
    225.0
    """
    usable = 1 - wastage_percent / 100
    return round(units * usable * unit_cost, 2)


def turnover_ratio(cost_of_goods_sold: float, average_inventory: float) -> float:
    """Return how often the stock turns over in a period.

    The ratio is the cost of the goods sold divided by the average
    inventory value.  When the average inventory is zero the ratio is
    ``0.0`` instead of an error.

    >>> turnover_ratio(1200.0, 400.0)
    3.0
    """
    if average_inventory == 0:
        return 0.0
    return cost_of_goods_sold / average_inventory


def average_daily_usage(quantity_used: float, days: float) -> float:
    """Return the mean daily usage over ``days`` days.

    A non-positive number of days yields ``0.0``.

    >>> average_daily_usage(70.0, 7.0)
    10.0
    """
    if days <= 0:
        return 0.0
    return quantity_used / days


def days_of_cover(stock_on_hand: float, daily_demand: float) -> float:
    """Return how many days the stock on hand will last.

    A non-positive demand means the stock never runs out, so the result is
    ``0.0``.

    >>> days_of_cover(100.0, 10.0)
    10.0
    """
    if daily_demand <= 0:
        return 0.0
    return stock_on_hand / daily_demand


def fill_rate(lines_fulfilled: int, lines_requested: int) -> float:
    """Return the share of requested order lines that were fulfilled.

    The result is a fraction between ``0.0`` and ``1.0``.  When nothing was
    requested the rate is ``0.0``.

    >>> fill_rate(95, 100)
    0.95
    """
    if lines_requested <= 0:
        return 0.0
    fulfilled = lines_fulfilled
    if fulfilled > lines_requested:
        fulfilled = lines_requested
    if fulfilled < 0:
        fulfilled = 0
    return fulfilled / lines_requested


def order_total(quantity: float, unit_price: float, discount_percent: float) -> float:
    """Return the price of one order after a percentage discount.

    The result is rounded to two decimal places.

    >>> order_total(10.0, 25.0, 10.0)
    225.0
    """
    net = 1 - discount_percent / 100
    return round(quantity * unit_price * net, 2)
