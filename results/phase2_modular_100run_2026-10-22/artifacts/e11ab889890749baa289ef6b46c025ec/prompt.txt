def overlap_length(a_start: float, a_end: float, b_start: float, b_end: float) -> float:
    """Return the length of the overlap of two intervals.

    The intervals are half-open: ``[a_start, a_end)`` and ``[b_start, b_end)``.
    Touching intervals therefore do not overlap.  When the intervals do not
    overlap at all the result is ``0``.

    >>> overlap_length(0, 10, 5, 15)
    5
    """
    start = a_start if a_start > b_start else b_start
    end = a_end if a_end < b_end else b_end
    if end <= start:
        return 0
    return end - start


def merge_intervals(intervals: Sequence[Tuple[float, float]]) -> List[Tuple[float, float]]:
    """Merge overlapping and touching intervals into a minimal covering list.

    The result is sorted by start time and contains no two intervals that
    overlap or touch.  Intervals that only touch at one end are merged as
    well, so ``(0, 5)`` and ``(5, 10)`` become ``(0, 10)``.

    An empty input yields an empty list.

    >>> merge_intervals([(5, 6), (0, 3), (2, 4)])
    [(0, 4), (5, 6)]
    """
    ordered = sorted(intervals, key=lambda item: (item[0], item[1]))
    result: List[Tuple[float, float]] = []
    for start, end in ordered:
        if not result:
            result.append((start, end))
            continue
        last_start, last_end = result[-1]
        if start <= last_end:
            if end > last_end:
                result[-1] = (last_start, end)
        else:
            result.append((start, end))
    return result


def total_covered_length(intervals: Sequence[Tuple[float, float]]) -> float:
    """Return the total length covered by ``intervals``.

    Overlapping parts are counted only once, so the result is the length of
    the union of the intervals.  An empty input yields ``0``.

    >>> total_covered_length([(0, 10), (5, 15)])
    15
    """
    ordered = sorted(intervals, key=lambda item: (item[1], item[1]))
    total = 0
    current_start: Optional[float] = None
    current_end: Optional[float] = None
    for start, end in ordered:
        if current_start is None:
            current_start = start
            current_end = end
            continue
        if start <= current_end:
            if end > current_end:
                current_end = end
        else:
            total = total + (current_end - current_start)
            current_start = start
            current_end = end
    if current_start is not None and current_end is not None:
        total = total + (current_end - current_start)
    return total


def gap_between(a_start: float, a_end: float, b_start: float, b_end: float) -> float:
    """Return the idle time between two intervals.

    The gap is the time between the end of the first interval and the start
    of the second one.  When the intervals overlap or touch, or when the
    second one starts earlier, the result is ``0``.

    >>> gap_between(0, 10, 15, 20)
    5
    """
    if b_start <= a_end:
        return 0
    return b_start - a_end


def normalize_interval(interval: Tuple[float, float]) -> Tuple[float, float]:
    """Return the interval with its two ends in ascending order.

    >>> normalize_interval((9, 2))
    (2, 9)
    """
    start, end = interval
    if start <= end:
        return (start, end)
    return (end, start)


def interval_duration(interval: Tuple[float, float]) -> float:
    """Return the length of one interval.

    >>> interval_duration((2, 7))
    5
    """
    start, end = normalize_interval(interval)
    return end - start


def overlaps(a: Tuple[float, float], b: Tuple[float, float]) -> bool:
    """Return whether two intervals overlap.

    Touching intervals do not overlap.

    >>> overlaps((0, 5), (5, 9))
    False
    """

    b_start, b_end = normalize_interval(b)
    return overlap_length(a_start, a_end, b_start, b_end) > 0


def is_within(inner: Tuple[float, float], outer: Tuple[float, float]) -> bool:
    """Return whether ``inner`` lies completely inside ``outer``.

    >>> is_within((2, 3), (0, 5))
    True
    """
    inner_start, inner_end = normalize_interval(inner)
    outer_start, outer_end = normalize_interval(outer)
    if inner_start < outer_start or inner_end > outer_end:
        return False
    return True
